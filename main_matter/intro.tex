\begin{algorithm}
  \caption{Euclides para calcular el MCD}\label{alg:euclides} % Título del algoritmo
  \begin{algorithmic}[1] % Numeración de líneas (opcional)
    \Require Dos números enteros positivos, \(a\) y \(b\)
    \Ensure El máximo común divisor de \(a\) y \(b\)
    \While{\(b \neq 0\)}
      \State \(r \gets a \mod b\)
      \State \(a \gets b\)
      \State \(b \gets r\)
    \EndWhile
    \State \Return \(a\)
  \end{algorithmic}
\end{algorithm}

\begin{lstlisting}[
  language=Swift,
  caption={Función recursiva para calcular el factorial de un número en Swift},
  label=lst:factorial_swift
]
func factorial(_ n: Int) -> Int {
    if n == 0 {
        return 1
    } else {
        return n * factorial(n - 1)
    }
}

// Ejemplo de uso
let numero = 5
let resultado = factorial(numero)
print("El factorial de \(numero) es \(resultado)")
\end{lstlisting}
